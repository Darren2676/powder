流程图 五、阶段二：待发货 → 生成发货申请
待发数量计算公式：
待发数量 = MAX(0, order_quantity - refunded_quantity - shipped_quantity)
建议发货量 = MAX(0, order_quantity - refunded_quantity - shipped_quantity - applied_quantity)

流程图 七、阶段四：退货管理流程
退货前提：
退货条件: 发货单状态 IN ('已发货', '已签收')
可退数量 = 发货批次数量 - ∑已退数量(排除已驳回) 
flowchart TB
    START["🔙 发起退货\n选择发货单"] --> LOAD["GET /api/return-orders/shipping-order/:sn\n加载发货单明细 + 批次\n计算各批次可退数量"]
    LOAD --> CHECK_STATUS{"发货单状态 = 已发货/已签收?"}
    CHECK_STATUS -- "否" --> ERR["❌ 不允许退货"]
    CHECK_STATUS -- "是" --> USER_INPUT["用户输入:\n退货类型 退款退货/退货换货\n退货仓库 / 退货原因\n每个批次退货数量"]
    USER_INPUT --> VALIDATE{"退货数量 ≤ 可退数量?"}
    VALIDATE -- "否" --> ERR2["❌ 超过可退数量"]
    VALIDATE -- "是" --> CREATE["POST /api/return-orders"]
    CREATE --> TX["开启事务"]
    TX --> INSERT_RO["插入 return_order 主表\nstatus = 待确认\napproval_status = 草稿"]
    INSERT_RO --> LOOP_D["🔁 遍历退货明细"]
    LOOP_D --> INSERT_ROD["插入 return_order_detail"]
    INSERT_ROD --> INSERT_BATCH["插入 return_order_batch"]
    INSERT_BATCH --> LOOP_D
    LOOP_D -- "完成" --> COMMIT["提交事务"]
    COMMIT --> RESULT["返回退货单号 RT-YYYYMMDD-NNN"]
    RESULT --> APPROVAL["提交审批"]
    APPROVAL --> APPROVE{"审批通过?"}
    APPROVE -- "是" --> CONFIRM["确认退货 POST /confirm"]
    APPROVE -- "否" --> REJECT["驳回退货 POST /reject"]
    CONFIRM --> INBOUND["退货入库 更新库存"]
    CONFIRM --> WRITEBACK["回写:\nsales_order_detail.refunded_quantity\nsales_order_detail.shipping_status"]

撤消 八、撤消发货申请 — 回写逻辑
flowchart TB
    CANCEL_SR["撤消发货申请"] --> CHECK{"当前状态?"}
    CHECK -- "待审核" --> DELETE["直接删除\nDELETE 主表+明细"]
    CHECK -- "已审核 未生成发货单" --> CANCEL["状态 → 已取消"]
    CHECK -- "已审核 已有发货单" --> ERR["❌ 已有发货单不可撤消\n需先撤消发货单"]
    DELETE --> WRITEBACK_D["回写 sales_order_detail:\n重新计算 shipping_status\n= 扣减本次申请量后的状态"]
    CANCEL --> WRITEBACK_C["回写 sales_order_detail:\n重新计算 shipping_status"]
    WRITEBACK_D --> DONE["✅ 完成"]
    WRITEBACK_C --> DONE
回写公式：
当前已申请总量 = ∑(所有非取消申请的 ship_quantity)
if (已申请总量 == 0) shipping_status = '未申请'
else if (已申请总量 < order_quantity) shipping_status = '部分发货'
else shipping_status = '全部发货'
撤消 八、撤消发货单 — 回写逻辑
flowchart TB
    CANCEL_SO["撤消发货单"] --> CHECK{"当前状态?"}
    CHECK -- "待发货" --> CANCEL1["状态 → 已取消"]
    CHECK -- "已发货 无退货" --> CANCEL2["状态 → 已取消"]
    CHECK -- "已发货 有退货" --> ERR1["❌ 已有退货单\n需先处理退货单"]
    CHECK -- "已签收" --> ERR2["❌ 已签收不可撤消"]
    CANCEL1 --> WRITEBACK_SR["回写发货申请:\n扣减已发数量\n重新判断申请状态"]
    CANCEL2 --> WRITEBACK_SR
    WRITEBACK_SR --> WRITEBACK_SOD["回写 sales_order_detail:\nshipped_quantity -= 本单发货量\n重新计算 shipping_status"]
    WRITEBACK_SOD --> DONE["✅ 完成"]
撤消 八、撤消/驳回退货单 — 回写逻辑
flowchart TB
    CANCEL_RT["撤消/驳回退货单"] --> CHECK{"当前状态?"}
    CHECK -- "待确认 草稿" --> DELETE["直接删除\nDELETE 主表+明细+批次"]
    CHECK -- "已审批未确认" --> REJECT["驳回: status → 已驳回"]
    CHECK -- "已确认" --> ERR["❌ 已确认不可撤消\n需要反向操作"]

    REJECT --> DONE2["✅ 驳回后不影响已发数据\n可退数量自动恢复 排除已驳回"]

关键不变式：
shipped_quantity = ∑(所有已确认发货单的发货数量)
refunded_quantity = ∑(所有已确认退货单的退货数量)
实际净发 = shipped_quantity - refunded_quantity
