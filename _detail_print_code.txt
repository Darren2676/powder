// ==================== 详细打印功能（三栏：基本信息+备料清单+工序任务） ====================
const handleDetailPrint = async () => {
  if (selectedRowKeys.value.length === 0) {
    message.warning('请先勾选需要打印的调度单')
    return
  }
  detailPrintLoading.value = true
  try {
    const res = await getPrintData(selectedRowKeys.value)
    if (res.success && res.data?.items?.length > 0) {
      detailPrintData.value = res.data.items
      setTimeout(() => {
        triggerDetailPrint()
      }, 300)
    } else {
      message.warning('未获取到打印数据')
    }
  } catch {
    message.error('获取打印数据失败')
  } finally {
    detailPrintLoading.value = false
  }
}

const triggerDetailPrint = () => {
  const items = detailPrintData.value
  if (!items || items.length === 0) return

  const printWindow = window.open('', '_blank', 'width=1000,height=750')
  if (!printWindow) {
    message.error('无法打开打印窗口，请检查浏览器是否允许弹出窗口')
    return
  }

  let pagesHtml = ''
  items.forEach((item: any, pageIdx: number) => {
    const order = item.order
    const materials = item.materials || []
    const tasks = item.tasks || []
    const scheduleName = getScheduleName(order.schedule_id)
    const isLast = pageIdx === items.length - 1

    const infoHtml = `
      <div class="section">
        <div class="section-title">一、生产单基本信息</div>
        <table class="info-table">
          <tr><td class="info-label">生产单编号</td><td class="info-value">${order.production_order_number || ''}</td><td class="info-label">生产计划编号</td><td class="info-value">${order.production_number || ''}</td><td class="info-label">生产日期</td><td class="info-value">${formatDate(order.production_date)}</td></tr>
          <tr><td class="info-label">产品编号</td><td class="info-value">${order.item_number || ''}</td><td class="info-label">产品名称</td><td class="info-value" colspan="3">${order.item_name || ''}</td></tr>
          <tr><td class="info-label">规格</td><td class="info-value">${order.specifications || ''}</td><td class="info-label">单位</td><td class="info-value">${order.basic_unit || ''}</td><td class="info-label">计划数量</td><td class="info-value">${order.planned_quantity ?? ''}</td></tr>
          <tr><td class="info-label">胶料编号</td><td class="info-value">${order.rubber_compound_number || ''}</td><td class="info-label">设备名称</td><td class="info-value">${order.equipment_name || order.equipment_number || ''}</td><td class="info-label">班次</td><td class="info-value">${scheduleName}</td></tr>
          <tr><td class="info-label">模具编号</td><td class="info-value">${order.mould_number || ''}</td><td class="info-label">成型件规格</td><td class="info-value">${order.formed_part_specifications || ''}</td><td class="info-label">成型件单耗</td><td class="info-value">${order.formed_part_unit_consumption || ''}</td></tr>
          <tr><td class="info-label">实际模腔</td><td class="info-value">${order.actual_cavity_count || ''}</td><td class="info-label">实际模穴</td><td class="info-value">${order.actual_hole_count || ''}</td><td class="info-label">实际班产</td><td class="info-value">${order.actual_daily_output || ''}</td></tr>
          <tr><td class="info-label">备注</td><td class="info-value" colspan="5">${order.remark || ''}</td></tr>
        </table>
      </div>`

    let materialsBody = ''
    if (materials.length > 0) {
      materials.forEach((m: any, mIdx: number) => {
        materialsBody += `<tr><td>${mIdx + 1}</td><td>${m.material_number || ''}</td><td class="text-left">${m.material_name || ''}</td><td>${m.material_type || ''}</td><td>${m.unit || ''}</td><td class="text-right">${m.required_quantity ?? ''}</td><td class="text-right">${m.adjusted_quantity ?? ''}</td><td class="blank-cell"></td><td>${m.step_number || ''}</td><td>${m.supply_type || ''}</td><td class="text-left">${m.remark || ''}</td></tr>`
      })
    } else {
      materialsBody = '<tr><td colspan="11" class="empty-note">暂无备料数据</td></tr>'
    }
    const materialsHtml = `
      <div class="section">
        <div class="section-title">二、备料清单</div>
        <table class="data-table">
          <thead><tr><th style="width:35px">序号</th><th style="width:100px">物料编号</th><th>物料名称</th><th style="width:55px">类型</th><th style="width:45px">单位</th><th style="width:70px">需求数量</th><th style="width:70px">调整数量</th><th style="width:80px">实发数量</th><th style="width:50px">工序</th><th style="width:65px">供应方式</th><th style="width:80px">备注</th></tr></thead>
          <tbody>${materialsBody}</tbody>
        </table>
      </div>`

    let tasksBody = ''
    if (tasks.length > 0) {
      tasks.forEach((t: any, tIdx: number) => {
        tasksBody += `<tr><td>${tIdx + 1}</td><td>${t.step_number || ''}</td><td class="text-left">${t.standard_process_name || ''}</td><td>${t.work_center_name || t.work_center_number || ''}</td><td class="text-right">${t.planned_quantity ?? ''}</td><td class="text-right">${t.completed_quantity ?? ''}</td><td class="blank-cell"></td><td>${t.task_status || ''}</td></tr>`
      })
    } else {
      tasksBody = '<tr><td colspan="8" class="empty-note">暂无工序任务数据</td></tr>'
    }
    const tasksHtml = `
      <div class="section">
        <div class="section-title">三、工序任务</div>
        <table class="data-table">
          <thead><tr><th style="width:35px">序号</th><th style="width:55px">工序号</th><th>工序名称</th><th style="width:90px">工作中心</th><th style="width:70px">计划数量</th><th style="width:70px">已完成</th><th style="width:80px">实际数量</th><th style="width:60px">状态</th></tr></thead>
          <tbody>${tasksBody}</tbody>
        </table>
      </div>`

    pagesHtml += `
      <div class="page${isLast ? ' last-page' : ''}">
        <div class="page-header-area"><div class="page-title">生产调度单</div><div class="page-subtitle">睿信橡胶密封件MES系统</div></div>
        <div class="page-meta"><span>第 ${pageIdx + 1} 页 / 共 ${items.length} 页</span><span>打印时间: ${dayjs().format('YYYY-MM-DD HH:mm')}</span></div>
        ${infoHtml}
        ${materialsHtml}
        ${tasksHtml}
        <div class="detail-footer"><div class="footer-item">调度员: <span class="sig-line"></span></div><div class="footer-item">班组长: <span class="sig-line"></span></div><div class="footer-item">操作员: <span class="sig-line"></span></div><div class="footer-item">日期: <span class="sig-line"></span></div></div>
      </div>`
  })

  const styles = `@page{size:A4 landscape;margin:8mm 10mm}*{margin:0;padding:0;box-sizing:border-box}body{font-family:"Microsoft YaHei","SimHei",sans-serif;color:#000;font-size:11px;-webkit-print-color-adjust:exact;print-color-adjust:exact}.page{page-break-after:always;padding:2mm 0}.page.last-page{page-break-after:avoid}.page-header-area{text-align:center;margin-bottom:4px}.page-title{font-size:18px;font-weight:bold;letter-spacing:4px}.page-subtitle{font-size:10px;color:#666;margin-top:1px}.page-meta{display:flex;justify-content:space-between;font-size:10px;color:#666;margin-bottom:6px;padding:0 2px}.section{margin-bottom:8px}.section-title{font-size:12px;font-weight:bold;margin-bottom:4px;padding-left:6px;border-left:3px solid #333}.info-table{width:100%;border-collapse:collapse}.info-table td{border:1px solid #000;padding:3px 6px;font-size:11px;line-height:1.4}.info-label{background-color:#f5f5f5;font-weight:bold;text-align:right;width:90px;white-space:nowrap}.info-value{min-width:100px}.data-table{width:100%;border-collapse:collapse}.data-table th,.data-table td{border:1px solid #000;padding:3px 5px;font-size:11px;text-align:center;line-height:1.3}.data-table th{background-color:#f0f0f0;font-weight:bold;white-space:nowrap}.data-table td.text-left{text-align:left}.data-table td.text-right{text-align:right}.data-table td.blank-cell{background-color:#ffffee;min-width:70px}.data-table td.empty-note{text-align:center;color:#999;padding:8px;font-style:italic}.detail-footer{display:flex;justify-content:space-between;margin-top:12px;padding:0 2px;font-size:11px}.footer-item{min-width:140px}.sig-line{display:inline-block;width:90px;border-bottom:1px solid #000;margin-left:6px}@media screen{body{padding:20px;background:#e8e8e8}.page{background:#fff;padding:20px 24px;margin:0 auto 20px;box-shadow:0 2px 8px rgba(0,0,0,.15);max-width:1100px}}`

  printWindow.document.write(`<!DOCTYPE html><html><head><meta charset="utf-8"><title>生产调度单 - 详细打印</title><style>${styles}</style></head><body>${pagesHtml}<script>window.onload=function(){setTimeout(function(){window.print()},400)};<\/script></body></html>`)
  printWindow.document.close()
}
